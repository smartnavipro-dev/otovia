## 🚀 Feature Design

### Feature 1: Adaptive Sharpening 🎯 **HIGH PRIORITY**

**Concept**: Analyze image characteristics and apply sharpening only when beneficial.

**Technical Approach**:
```kotlin
fun shouldApplySharpening(bitmap: Bitmap): Boolean {
    // Calculate image contrast using OpenCV
    val mat = Mat()
    Utils.bitmapToMat(bitmap, mat)

    // Convert to grayscale
    val gray = Mat()
    Imgproc.cvtColor(mat, gray, Imgproc.COLOR_RGBA2GRAY)

    // Calculate standard deviation (contrast measure)
    val mean = MatOfDouble()
    val stdDev = MatOfDouble()
    Core.meanStdDev(gray, mean, stdDev)

    val contrast = stdDev.get(0, 0)[0]

    // High contrast images (>50) may not benefit from sharpening
    // Low contrast images (<30) need sharpening
    return contrast < 50.0
}

private fun applyAdaptivePreprocessing(bitmap: Bitmap): Bitmap {
    // Stage 1: 2x scaling
    val scaled2x = Bitmap.createScaledBitmap(bitmap, width * 2, height * 2, true)

    // Stage 2: Contrast enhancement
    val contrastBitmap = applyContrastEnhancement(scaled2x)

    // Stage 3: Adaptive sharpening
    val processedBitmap = if (shouldApplySharpening(contrastBitmap)) {
        Log.d(TAG, "Applying sharpening (low contrast detected)")
        applySharpenFast(contrastBitmap)
    } else {
        Log.d(TAG, "Skipping sharpening (high contrast)")
        contrastBitmap
    }

    // Stage 4: Final 2x scaling
    return Bitmap.createScaledBitmap(processedBitmap, width * 4, height * 4, true)
}
```

**Expected Benefits**:
- **Performance**: 20-30ms reduction on high-contrast images (50% skip rate)
- **Average Time**: 437ms → 410ms (6% improvement)
- **OCR Accuracy**: Maintained or improved (avoid over-sharpening artifacts)

**Implementation Time**: 2-4 hours

```kotlin
enum class SharpenStrength(val multiplier: Float) {
    OFF(0.0f),
    LIGHT(0.5f),
    NORMAL(1.0f),
    STRONG(1.5f),
    MAXIMUM(2.0f)
}

fun applySharpenFastWithStrength(bitmap: Bitmap, strength: SharpenStrength): Bitmap {
    if (strength == SharpenStrength.OFF) {
        return bitmap
    }

    val mat = Mat()
    Utils.bitmapToMat(bitmap, mat)

    val sharpened = Mat()
    val kernel = Mat(3, 3, CvType.CV_32F).apply {
        val alpha = strength.multiplier
        put(0, 0,
            0.0, -alpha, 0.0,
            -alpha, 1.0 + 4.0 * alpha, -alpha,
            0.0, -alpha, 0.0
        )
    }

    Imgproc.filter2D(mat, sharpened, -1, kernel)

    val result = Bitmap.createBitmap(bitmap.width, bitmap.height, Bitmap.Config.ARGB_8888)
    Utils.matToBitmap(sharpened, result)

    return result
}
```

**Expected Benefits**:
- **User Control**: Adapt to different content types
- **OCR Accuracy**: 53-55% with optimal strength selection
- **Performance**: Light mode (0.5x) → ~80ms, Maximum (2.0x) → ~120ms

**Implementation Time**: 4-6 hours

**Implementation**:
```kotlin
data class ImageQualityMetrics(
    val contrast: Int,      // 0-100
    val sharpness: Int,     // 0-100
    val brightness: Int     // 0-100
) {
    val overallQuality: String
        get() = when {
            contrast > 70 && sharpness > 60 && brightness in 40..60 -> "Excellent ✓"
            contrast > 50 && sharpness > 40 -> "Good ✓"
            contrast < 30 || sharpness < 20 -> "Poor ⚠"
            else -> "Fair"
        }

    val recommendation: String?
        get() = when {
            contrast < 30 -> "Try increasing screen brightness"
            brightness > 80 -> "Screen too bright, reduce brightness"
            brightness < 20 -> "Screen too dark, increase brightness"
            sharpness < 30 -> "Consider using Strong sharpening"
            else -> null
        }
}

fun calculateImageQuality(bitmap: Bitmap): ImageQualityMetrics {
    val mat = Mat()
    Utils.bitmapToMat(bitmap, mat)

    val gray = Mat()
    Imgproc.cvtColor(mat, gray, Imgproc.COLOR_RGBA2GRAY)

    // Contrast: Standard deviation normalized to 0-100
    val mean = MatOfDouble()
    val stdDev = MatOfDouble()
    Core.meanStdDev(gray, mean, stdDev)
    val contrast = (stdDev.get(0, 0)[0] * 100 / 64).toInt().coerceIn(0, 100)

    // Sharpness: Laplacian variance
    val laplacian = Mat()
    Imgproc.Laplacian(gray, laplacian, CvType.CV_64F)
    val sharpnessVar = MatOfDouble()
    Core.meanStdDev(laplacian, MatOfDouble(), sharpnessVar)
    val sharpness = (sharpnessVar.get(0, 0)[0] / 50).toInt().coerceIn(0, 100)

    // Brightness: Mean intensity
    val brightness = (mean.get(0, 0)[0] * 100 / 255).toInt()

    return ImageQualityMetrics(contrast, sharpness, brightness)
}
```

**Expected Benefits**:
- **User Education**: Help users understand what affects OCR quality
- **Debugging**: Identify problematic content
- **Optimization**: Guide sharpening strength selection

**Implementation Time**: 6-8 hours

### Feature 4: A/B Testing Framework 🎯 **LOW PRIORITY**

**Concept**: Automatically test multiple preprocessing pipelines and select the best.

**Implementation** (Future - v1.2.0):
```kotlin
data class PreprocessingPipeline(
    val name: String,
    val sharpenStrength: SharpenStrength,
    val contrastMultiplier: Float,
    val scalingStrategy: ScalingStrategy
)

suspend fun selectBestPipeline(bitmap: Bitmap): PreprocessingPipeline {
    val pipelines = listOf(
        PreprocessingPipeline("Fast", SharpenStrength.LIGHT, 1.0f, ScalingStrategy.TWO_STAGE),
        PreprocessingPipeline("Balanced", SharpenStrength.NORMAL, 1.0f, ScalingStrategy.TWO_STAGE),
        PreprocessingPipeline("Accuracy", SharpenStrength.STRONG, 1.2f, ScalingStrategy.TWO_STAGE)
    )

    // Test each pipeline on a small sample
    val results = pipelines.map { pipeline ->
        val processed = applyPipeline(bitmap, pipeline)
        val quality = calculateImageQuality(processed)
        pipeline to quality.overallQuality
    }

    // Select pipeline with best quality
    return results.maxByOrNull { it.second }?.first ?: pipelines[1]
}
```

**Expected Benefits**:
- **Automatic Optimization**: Best pipeline per content type
- **Future Extensibility**: Framework for adding new preprocessing methods
